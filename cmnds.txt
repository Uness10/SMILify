# =============================================================================
# JOINT-LIMIT PRIOR STUDY — lambda sweep + qualitative renders
# RWTH CLAIX-2023, account rwth2151 (MaxWall 24 h, MaxSubmit 1000, MaxJobs 800)
#
# Design changed from the previous run:
#   - constrained arms now train 50 additional epochs (was 10)
#   - one arm per lambda in {1e-4, 1e-3, 1e-2, 1e-1}, SINGLE-VIEW only
#   - renders of the SAME exported-pose windows through every arm, for
#     qualitative comparison (from clip_*.npz, no re-inference)
#
# Run everything from the repo root on $HPCWORK.
# =============================================================================

cd "$HPCWORK/SMILify"
conda activate "$HPCWORK/conda_envs/pytorch3d"
mkdir -p logs configs_runs

export ENV_PREFIX=$HPCWORK/conda_envs/pytorch3d
export ACCT=rwth2151

# The single-view checkpoint the whole study branches from, and the JSON that
# PRODUCED it. Do not let prepare_resume_config.py fall back to the checkpoint's
# embedded config — it carries stale defaults.
export SV_REF=<path to the downloaded single-view .pth>
export SV_BASE_CONFIG=<the JSON that produced it>


# -----------------------------------------------------------------------------
# 0. SANITY CHECK FIRST — is there anything for the prior to fix?
# -----------------------------------------------------------------------------
# Score the reference on its own. It needs no training, so it runs immediately
# and in parallel with everything else. Its violation table is the single most
# informative number in the study: if the reference is already inside the
# authored ranges, the hinge is near-zero by construction at EVERY lambda and
# the sweep will find nothing. That answer costs one GPU-hour, not twenty.

sbatch --array=0 --account=$ACCT --export=ALL,SV_REF,ENV_PREFIX \
    hpc_files/rwth/run_prior_study_eval.sbatch

# When it lands:
#   column -s, -t prior_study_results/sv_reference/analysis/limit_violations.csv | head -20


# -----------------------------------------------------------------------------
# 1. WALL-CLOCK CHECK — will 50 epochs fit in 24 h?
# -----------------------------------------------------------------------------
# 24 h is the account cap; there is no longer QOS available on this association.
# The previous +10-epoch runs were budgeted at 12 h. Find the per-epoch time in
# their logs and multiply by 50 BEFORE submitting four of these.

grep -h "Epoch\|epoch" logs/jl_train_*_*.out | grep -i "time\|elapsed\|completed" | tail -20

# If 50 epochs does not fit, in order of preference:
#   a) lower EXTRA_EPOCHS — the sweep is about lambda, and the comparison holds
#      as long as every arm gets the SAME count;
#   b) drop dataset.dataset_fraction for the sweep, restore it for the winner;
#   c) chain a continuation job with --dependency=afterany and re-run
#      prepare_resume_config.py against the newest checkpoint_epoch_*.pth.
# The prep script sets save_checkpoint_every=2 so (c) costs at most 2 epochs.


# -----------------------------------------------------------------------------
# 2. BUILD THE FOUR SWEEP CONFIGS (login node)
# -----------------------------------------------------------------------------
# One call per lambda, all with the same --extra-epochs. Writes
# configs_runs/singleview_lam{1e-4,1e-3,1e-2,1e-1}.json — exactly the names the
# training array derives from its task id.

EXTRA_EPOCHS=50 bash scripts/prior_study/prepare_lambda_sweep.sh singleview

# Check the arithmetic it prints for EVERY lambda before continuing:
#   [prepare] resume from epoch N+1, train 50 epoch(s) -> training.num_epochs = N+51
#   [prepare] arm: CONSTRAINED (w=<lambda>)
#   [patch]   save_checkpoint_every: 5 -> 2
#
# num_epochs is an ABSOLUTE END epoch, not a count. A checkpoint at epoch 354
# with num_epochs=50 trains ZERO epochs and exits looking successful — which is
# why "50 epochs" is passed as --extra-epochs and never typed into the JSON.

grep -H '"num_epochs"\|joint_limit_regularization' configs_runs/singleview_lam*.json


# -----------------------------------------------------------------------------
# 3. PRE-FLIGHT (login node) — do not skip
# -----------------------------------------------------------------------------
# Catches the two failures that do NOT crash: architecture drift (which resets
# the epoch counter to 0, turning "50 epochs" into 400 from scratch) and test
# splits that disagree between arms (which scores them on different frames).

python scripts/prior_study/preflight_study.py \
    --config configs_runs/singleview_lam1e-4.json --reference "$SV_REF" \
    --config configs_runs/singleview_lam1e-3.json --reference "$SV_REF" \
    --config configs_runs/singleview_lam1e-2.json --reference "$SV_REF" \
    --config configs_runs/singleview_lam1e-1.json --reference "$SV_REF"


# -----------------------------------------------------------------------------
# 4. TRAIN THE SWEEP — 4 tasks, one whole 4xH100 node each
# -----------------------------------------------------------------------------
# %2 throttles to two concurrent tasks. Four full nodes at once is within the
# account limits (MaxJobs 800), but the queue wait for two is usually shorter
# than for four, and a mistake caught on task 0 costs half as much.

jid=$(sbatch --parsable --array=0-3%2 --account=$ACCT --export=ALL,ENV_PREFIX \
        hpc_files/rwth/run_prior_study_train.sbatch)
echo "training array: $jid"

# One lambda only:
#   sbatch --array=1 --account=$ACCT --export=ALL,ENV_PREFIX hpc_files/rwth/run_prior_study_train.sbatch
# The multi-view sweep later, at whichever lambda wins:
#   sbatch --array=4-7 --account=$ACCT --export=ALL,ENV_PREFIX hpc_files/rwth/run_prior_study_train.sbatch


# -----------------------------------------------------------------------------
# 5. SCORE THE SWEEP
# -----------------------------------------------------------------------------
# Tasks 1-4 = the four lambdas (task 0, the reference, already ran in step 0).
# afterany rather than afterok: a training task killed at the 24 h wall still
# left periodic checkpoints worth scoring, and run_prior_study.sh falls back to
# the newest one when best_model.pth was never written.

sbatch --array=1-4 --dependency=afterany:$jid --account=$ACCT \
    --export=ALL,SV_REF,ENV_PREFIX,EXTRA_EPOCHS=50 \
    hpc_files/rwth/run_prior_study_eval.sbatch


# -----------------------------------------------------------------------------
# 6. QUALITATIVE RENDERS — from the exported poses, not from raw video
# -----------------------------------------------------------------------------
# The eval stage already wrote each arm's predictions to
#   prior_study_results/<arm>/clip_*.npz   (+ the .json sidecar already in git)
# Those ARE the frames the study scored. Rendering them means the pictures and
# the tables describe the same predictions, and it costs minutes on one GPU —
# no checkpoint, no dataset, no backbone.

# --- 6a. Pick the windows ONCE (login node, CPU, seconds).
# Every arm must render the SAME frames or the grid is not a comparison.
# --select worst aims them at the windows where the REFERENCE violates the
# authored limits most — the frames the prior is supposed to fix.

python scripts/prior_study/pick_segments.py \
    --npz prior_study_results/sv_reference/clip_sv_reference.npz \
    --smal-file 3D_model_prep/SMILy_STICK_limits_authored.pkl \
    --segments 3 --length 300 --select worst \
    --out prior_study_results/renders/segments.json

# Read its output before going further. If it reports that the reference never
# leaves the authored ranges, no render can show the prior correcting anything
# and the sweep's answer is already in — see step 0.
#
# Add an unbiased set alongside the worst-case one:
#   python scripts/prior_study/pick_segments.py ... --select even \
#       --out prior_study_results/renders/segments_even.json

# --- 6b. Render every arm over those windows (1 GPU each, ~minutes).
sbatch --array=0-4 --account=$ACCT --export=ALL,ENV_PREFIX \
    hpc_files/rwth/run_prior_study_render.sbatch

# Reference alone, as soon as its eval has landed:
#   sbatch --array=0 --account=$ACCT --export=ALL,ENV_PREFIX \
#       hpc_files/rwth/run_prior_study_render.sbatch
#
# Default camera is `orbit`: root-centred, so the predicted translation is
# dropped and only ARTICULATION varies between panels — which is what a joint
# limit changes, and what makes two arms comparable frame by frame.
# For the model's own view (checks the fit still lands on the animal):
#   CAMERA=predicted sbatch --array=0-4 --account=$ACCT \
#       --export=ALL,ENV_PREFIX,CAMERA hpc_files/rwth/run_prior_study_render.sbatch

# --- 6c. Stack into one comparison video per window (login node, ffmpeg).
bash scripts/prior_study/stack_renders.sh
#   -> prior_study_results/renders/comparison/<segment>_compare.mp4
#      panels: reference | 1e-4 | 1e-3 | 1e-2 | 1e-1, identical frames
#
# Each panel carries a HUD: arm, frame index, and how many of the 129 authored
# joint-axes are out of range IN THAT FRAME. Watch that counter fall across
# panels — it is the per-frame version of the number in limit_violations.csv.

# Just two arms side by side (easier to judge than five):
#   ARMS="sv_reference lam1e-2" bash scripts/prior_study/stack_renders.sh


# -----------------------------------------------------------------------------
# 7. READ THE RESULTS
# -----------------------------------------------------------------------------
#   prior_study_results/sweep/sweep.md          <- the lambda curve (HEADLINE)
#   prior_study_results/sweep/sweep.csv
#   prior_study_results/lam<L>/comparison/comparison.md   <- reference vs that lambda
#   prior_study_results/renders/comparison/*_compare.mp4  <- the qualitative side
#   prior_study_results/renders/<arm>/render.json         <- per-segment mean
#                                                            axes-out-of-range
#
# Win condition: a lambda that lowers the violation rate while leaving MPJPE and
# PCK flat. A lambda that lowers violations AND accuracy is the prior overriding
# the data. Flat everywhere, with a flat reference violation table, means the
# model was already inside the authored ranges — see step 0.
#
# CAVEAT to state alongside every number: the reference arm received ZERO extra
# epochs, so reference-vs-lambda mixes the prior with 50 epochs of continued
# training. Comparisons BETWEEN lambdas are clean — they share the epoch count.
# The control that would separate them is one more prepare_lambda_sweep.sh call
# with LAMBDAS="0", which is four lines of work if a reviewer asks.
